#include <stdio.h>

int main()
{
    int N, counter = 0;
    
    printf("Enter numbers, 0 to end:");
    scanf("%d", &N);
    
    while(N != 0) {
        if( N % 2 == 0) {
            counter = counter + 1;
        }
        scanf("%d", &N);
    }
    printf("%d\n", counter);
    return 0;
}
